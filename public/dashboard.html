// ========================================================= // LOAD DASHBOARD - IMPROVED VERSION // ========================================================= async function loadDashboard() { try { const user = authManager.getUser(); if (!user) { showToast('error', 'Please login again'); window.location.href = CONFIG.ROUTES.LOGIN; return; } // Get member profile - handle gracefully if not found const memberResult = await supabaseClient.getMember(user.id); if (!memberResult.success) { console.warn('Member not found:', memberResult.error); // Check if this is a new user without member record if (memberResult.error && memberResult.error.includes('PGRST116')) { showToast('info', 'Please complete your registration.'); // Redirect to registration or show a message document.getElementById('loadingScreen').style.display = 'none'; document.getElementById('dashboardContent').innerHTML = `

Welcome to Masika Benevolent!

Please complete your registration to access your dashboard.

Complete Registration
`; return; } showToast('error', 'Failed to load member profile'); return; } const member = memberResult.data; if (!member) { // No member record found document.getElementById('loadingScreen').style.display = 'none'; document.getElementById('dashboardContent').innerHTML = `

Complete Your Registration

You are logged in but need to complete your membership registration.

Complete Registration
`; return; } // Store member in localStorage localStorage.setItem(CONFIG.STORAGE.MEMBER, JSON.stringify(member)); // Render dashboard renderDashboard(member); // Load dependants const depsResult = await supabaseClient.getMemberDependants(member.id); if (depsResult.success) { renderFamilyMembers(depsResult.data || []); document.getElementById('dependantCount').textContent = (depsResult.data || []).length; } // Load payments const paymentsResult = await supabaseClient.getMemberPayments( member.membership_number, 5 ); if (paymentsResult.success) { renderPayments(paymentsResult.data || []); } // Show content document.getElementById('loadingScreen').style.display = 'none'; document.getElementById('dashboardContent').style.display = 'block'; } catch (error) { console.error('Dashboard load error:', error); showToast('error', 'Failed to load dashboard'); document.getElementById('loadingScreen').style.display = 'none'; } }
